注解配置文件

```java
@Configuration
@ComponentScan(basePackages = {"com.weibo"})//注意传入的数组要加{}
public class JavaConfig {
    @Bean("user")//为即User的id，如果没写为使用方法名
    @DependsOn({"cat"})//使用dependsOn表示再cat对象注入后，User再注入
    public User user(Cat cat){
        User user = new User();
        user.setCat(cat);
        return user;
    }

    @Bean("cat")
    public Cat cat(){
        Cat cat = new Cat();
        return cat;
    }
}
/*
因为两个类是循环依赖的，如当test调用User时必须给User的car属性赋值，因为User会有空指针错误，而调用Cat会显示为null;
如果想正常运行需要：去掉Cat打印User对象
*/
```

```java
@Component
public class Cat {
    @Value("小花")//使用Value属性为属性注入value，此属性无默认值，需赋值String
    private String name;
    @Value("1")
    private Integer age;

@Component
public class User {
    private String name;
    private Integer age;
    private BigInteger id;
    private Cat cat;
    //getter and setter
    //consturctor
    @Override
    public String toString() {
        final StringBuffer sb = new StringBuffer("user{");
        sb.append("name='").append(name).append('\'');
        sb.append(", age=").append(age);
        sb.append(", id=").append(id);
        sb.append(", cat={猫的名字: ").append(cat.get().getName()+" ,猫的年龄: "+cat.orElse(new Cat()).getAge());
        sb.append('}');
        return sb.toString();
    }
}
```
